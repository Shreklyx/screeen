# Screen Share – the native apps (no Electron, no Chromium).
#   core/       C++17, every platform: direct mode's transport, crypto, signaling, the session engine
#   windows/    the Windows app (Win32, Direct3D 11, Media Foundation, NVENC, WASAPI)
#   mac/        the macOS app (AppKit, ScreenCaptureKit, VideoToolbox, Core Audio)
#   tests/      core tests and a command-line host/viewer (Linux, CI)
cmake_minimum_required(VERSION 3.20)
project(ScreenShareNative LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>") # no VC++ runtime to install
endif()

add_library(sscore STATIC
  core/clock.cpp core/crypto.cpp core/json.cpp core/net.cpp core/ws_plain.cpp core/ws_bench.cpp core/direct.cpp core/engine.cpp)
target_include_directories(sscore PUBLIC core)
if(WIN32)
  target_link_libraries(sscore PUBLIC ws2_32 iphlpapi bcrypt)
  target_compile_definitions(sscore PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
elseif(APPLE)
  target_link_libraries(sscore PUBLIC "-framework Security")
else()
  find_package(Threads REQUIRED)
  target_link_libraries(sscore PUBLIC Threads::Threads)
endif()
if(NOT MSVC)
  target_compile_options(sscore PRIVATE -Wall -Wextra)
endif()

option(SS_TESTS "core tests and the command-line host/viewer" ON)
if(SS_TESTS)
  add_executable(sscore_test tests/core_test.cpp)
  target_link_libraries(sscore_test PRIVATE sscore)
  # our own cryptography against the standards' test vectors: on this CPU's AES instructions, and once more
  # in the portable code (what a CPU without them runs)
  add_executable(crypto_test tests/crypto_test.cpp)
  target_link_libraries(crypto_test PRIVATE sscore)
  add_executable(crypto_portable_test tests/crypto_test.cpp core/crypto.cpp)
  target_include_directories(crypto_portable_test PRIVATE core)
  target_compile_definitions(crypto_portable_test PRIVATE SS_CRYPTO_PORTABLE)
  if(WIN32)
    target_link_libraries(crypto_portable_test PRIVATE bcrypt)
    target_compile_definitions(crypto_portable_test PRIVATE _CRT_SECURE_NO_WARNINGS)
  elseif(APPLE)
    target_link_libraries(crypto_portable_test PRIVATE "-framework Security")
  endif()
  # lossless sound (FLAC, our own encoder and decoder)
  add_executable(lossless_test tests/lossless_test.cpp common/lossless.cpp)
  target_include_directories(lossless_test PRIVATE common)
  # spatial sound (our own head model)
  add_executable(spatial_test tests/spatial_test.cpp common/spatial.cpp)
  target_include_directories(spatial_test PRIVATE common)
  add_executable(ss-cli tests/ss_cli.cpp common/lossless.cpp)
  target_include_directories(ss-cli PRIVATE common)
  target_link_libraries(ss-cli PRIVATE sscore)
  # the viewers' sound buffer (lossless only – our own FLAC code, no library)
  add_executable(audio_ring_test tests/audio_ring_test.cpp common/audio_ring.cpp common/lossless.cpp common/spatial.cpp)
  target_include_directories(audio_ring_test PRIVATE common)
  if(TARGET Threads::Threads) # (Linux; Windows and macOS have threads in their system libraries)
    target_link_libraries(audio_ring_test PRIVATE Threads::Threads)
  endif()
endif()

if(WIN32 AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/windows/CMakeLists.txt)
  add_subdirectory(windows)
endif()
if(APPLE AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/mac/CMakeLists.txt)
  add_subdirectory(mac)
endif()
